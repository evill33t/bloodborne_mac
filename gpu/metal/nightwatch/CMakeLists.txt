# Peer compatibility follows this fork's actual selected source, including adapters.
file(GLOB_RECURSE NW_ID_SOURCES CONFIGURE_DEPENDS "${NW_VENDOR}/*.cpp" "${NW_VENDOR}/*.h" "${NW_VENDOR}/*.inc" "${NW_VENDOR}/*.xml" "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.inc")
list(APPEND NW_ID_SOURCES "${CMAKE_CURRENT_LIST_FILE}" "${BB_ROOT}/scripts/macos_nightwatch.py" "${BB_ROOT}/scripts/macos_prepare.py")
list(SORT NW_ID_SOURCES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${NW_ID_SOURCES})
set(NW_ID_CONTENT "")
foreach(path IN LISTS NW_ID_SOURCES)
    file(SHA256 "${path}" digest)
    string(APPEND NW_ID_CONTENT "${digest}")
endforeach()
string(SHA256 NW_BUILD_ID "${NW_ID_CONTENT}")
string(SUBSTRING "${NW_BUILD_ID}" 0 16 NW_BUILD_ID)
set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/darwin.cpp" PROPERTIES COMPILE_DEFINITIONS BB_NIGHTWATCH_BUILD_ID="bloodborne-mac-nightwatch-${NW_BUILD_ID}")
